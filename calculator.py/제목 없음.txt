
def add(a, b):
    return a + b

def subtract(a, b):
    return a - b

def multiply(a, b):
    return a * b

def divide(a, b):
    if b == 0:
        return "0으로 나눌 수 없습니다."
    return a / b

print("나의 첫 번째 계산기")
print("1. 더하기")
print("2. 빼기")
print("3. 곱하기")
print("4. 나누기")

choice = input("번호 선택: ")
a = float(input("첫 번째 숫자: "))
b = float(input("두 번째 숫자: "))

if choice == "1":
    result = add(a, b)
elif choice == "2":
    result = subtract(a, b)
elif choice == "3":
    result = multiply(a, b)
elif choice == "4":
    result = divide(a, b)
else:
    result = "잘못된 선택"

print("결과:", result)
