
# BMI 계산 프로그램
# 키와 몸무게를 이용한 체질량지수 계산

print("===== BMI 계산기 =====")

height = float(input("키를 입력하세요 (cm): "))
weight = float(input("몸무게를 입력하세요 (kg): "))

height_m = height / 100

bmi = weight / (height_m ** 2)

print("\n===== 계산 결과 =====")
print("BMI:", round(bmi, 2))

if bmi < 18.5:
    print("결과: 저체중")
elif bmi < 25:
    print("결과: 정상")
elif bmi < 30:
    print("결과: 과체중")
else:
    print("결과: 비만")
